cmake_minimum_required(VERSION 3.25)
project(FloraGPA VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "FloraGPA requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8 /EHsc)
endif()
option(FLORA_BUILD_GUI "Build the Qt Widgets application" ON)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui)
set(CMAKE_AUTOMOC ON)
add_library(FloraCore src/core/Frame.cpp src/core/Hash.cpp)
target_link_libraries(FloraCore PUBLIC bcrypt)
target_include_directories(FloraCore PUBLIC src)
add_library(FloraReplay src/replay/Replay.cpp src/replay/TexturePreview.cpp)
target_link_libraries(FloraReplay PUBLIC FloraCore d3d11 dxgi dxguid d3dcompiler)
add_library(FloraExperiment src/application/Experiment.cpp src/application/ShaderInspector.cpp)
target_include_directories(FloraExperiment SYSTEM PUBLIC third_party)
target_link_libraries(FloraExperiment PUBLIC FloraReplay Qt6::Core)
foreach(name FloraGPA.Cli FloraGPA.Worker)
  add_executable(${name} src/cli/main.cpp)
  target_link_libraries(${name} PRIVATE FloraExperiment Qt6::Core Qt6::Gui psapi)
endforeach()
if(FLORA_BUILD_GUI)
  find_package(Qt6 6.11 REQUIRED COMPONENTS Widgets Concurrent)
  add_library(FloraUi src/app/Models.cpp src/app/Models.h src/app/Views.cpp src/app/Views.h src/app/MainWindow.cpp src/app/MainWindow.h)
  target_link_libraries(FloraUi PUBLIC FloraExperiment Qt6::Widgets Qt6::Concurrent)
  add_executable(FloraGPA WIN32 src/app/main.cpp)
  target_link_libraries(FloraGPA PRIVATE FloraUi)
  add_dependencies(FloraGPA FloraGPA.Worker)
  if(BUILD_TESTING)
    find_package(Qt6 6.11 REQUIRED COMPONENTS Test)
    set(FLORA_TEST_CAPTURE_DIR "" CACHE PATH "External GF2/BF1 fixture directory")
    add_executable(FloraUiTests tests/UiTests.cpp)
    target_link_libraries(FloraUiTests PRIVATE FloraUi Qt6::Test)
    add_dependencies(FloraUiTests FloraGPA.Worker)
    add_test(NAME ui COMMAND FloraUiTests -o -,txt)
    set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;FLORA_TEST_CAPTURE_DIR=${FLORA_TEST_CAPTURE_DIR}" ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
    add_executable(FloraExperimentTests tests/ExperimentTests.cpp)
    target_link_libraries(FloraExperimentTests PRIVATE FloraExperiment Qt6::Test)
    add_test(NAME experiments COMMAND FloraExperimentTests -o -,txt)
    set_tests_properties(experiments PROPERTIES ENVIRONMENT "FLORA_TEST_CAPTURE_DIR=${FLORA_TEST_CAPTURE_DIR}" ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
    add_executable(FloraShaderTests tests/ShaderInspectorTests.cpp)
    target_link_libraries(FloraShaderTests PRIVATE FloraExperiment Qt6::Test)
    add_test(NAME shaders COMMAND FloraShaderTests -o -,txt)
    set_tests_properties(shaders PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
  endif()
endif()
if(BUILD_TESTING)
  add_executable(FloraCoreTests tests/CoreTests.cpp)
  target_link_libraries(FloraCoreTests PRIVATE FloraCore)
  add_test(NAME core COMMAND FloraCoreTests)
endif()
