cmake_minimum_required(VERSION 3.25)
project(FloraNativeDrawFixture LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The native draw fixture requires Windows x64")
endif()
add_executable(FloraNativeDrawFixture ../native_draw_fixture.cpp)
target_compile_features(FloraNativeDrawFixture PRIVATE cxx_std_20)
target_include_directories(FloraNativeDrawFixture SYSTEM PRIVATE ../../../third_party)
target_link_libraries(FloraNativeDrawFixture PRIVATE d3d11 dxgi d3dcompiler bcrypt psapi)
if(MSVC)
  target_compile_options(FloraNativeDrawFixture PRIVATE /W4 /permissive- /utf-8 /EHsc)
endif()
